📊 구글 스프레드시트 연동 설정

스프레드시트 공유 링크(docs.google.com) 또는 Apps Script 웹 앱 URL(script.google.com)을 입력하세요.

💡 연동 방식 선택 안내:
방법 1. 구글 시트 공유 링크 (가장 간편한 불러오기)

스프레드시트 우측 상단 [공유] 클릭 > 일반 액세스를 '링크가 있는 모든 사용자 - 뷰어'로 설정한 후 브라우저 주소창의 시트 주소를 위 입력창에 그대로 넣으시면 바로 데이터를 읽어옵니다.

※ 시트 구성: A열(학생명), B열(역할명), C열(역할설명) (제목 행 포함 가능)

방법 2. Apps Script 웹 앱 (불러오기 + 자동 시트 저장)
  1. 구글 시트 메뉴에서 확장 프로그램 > Apps Script 클릭
  2. 아래 코드를 붙여넣고 [배포] > [새 배포] 클릭
  3. 유형: 웹 앱, 액세스 권한: 모든 사용자(Anyone) 설정 후 배포
  4. 발급된 /exec URL을 위 입력창에 등록
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var values = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", data: values }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet();
    var target = sheet.getSheetByName("배정결과") || sheet.insertSheet("배정결과");
    target.clear();
    target.appendRow(["번호", "아이콘", "역할명", "학생이름", "역할설명", "배정일시"]);
    var list = contents.assignments || [];
    for (var i = 0; i < list.length; i++) {
      var item = list[i];
      target.appendRow([i + 1, item.roleIcon, item.roleTitle, item.studentName, item.roleDesc, contents.timestamp]);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "saved" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", msg: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
